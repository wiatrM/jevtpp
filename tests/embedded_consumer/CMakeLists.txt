cmake_minimum_required(VERSION 3.24)
project(jevt_embedded_consumer LANGUAGES CXX)

if(NOT JEVT_SOURCE_DIR)
  message(FATAL_ERROR "Pass JEVT_SOURCE_DIR pointing to the JevT++ source tree")
endif()
option(EXPECT_JEVT_EXAMPLES "Expect an explicitly enabled JevT++ example" OFF)
add_subdirectory("${JEVT_SOURCE_DIR}" jevtpp)
if(JEVT_BUILD_TESTS OR JEVT_BUILD_BENCHMARKS)
  message(FATAL_ERROR "Embedded defaults must not enable tests or benchmarks")
endif()
if(EXPECT_JEVT_EXAMPLES)
  if(NOT JEVT_BUILD_EXAMPLES OR NOT TARGET jevt_support_routing)
    message(FATAL_ERROR "Explicit example option was not preserved")
  endif()
elseif(JEVT_BUILD_EXAMPLES)
  message(FATAL_ERROR "Embedded defaults must not enable examples")
endif()
if(TARGET jevt_acceptance_tests OR TARGET jevt_benchmark OR TARGET jevt_diagnostics_contention)
  message(FATAL_ERROR "Unexpected embedded test or benchmark targets")
endif()
add_executable(embedded_consumer ../consumer/main.cpp)
target_link_libraries(embedded_consumer PRIVATE jevt::jevt)
enable_testing()
add_test(NAME embedded_consumer COMMAND embedded_consumer)
